// 1. Binary Max-Heap Priority Queue for Critical Emergencies (P1/P2)
class BinaryMaxHeapPriorityQueue {
  constructor() {
    this.heap = [];
  }

  enqueue(item) {
    this.heap.push(item);
    this.bubbleUp(this.heap.length - 1);
  }

  bubbleUp(index) {
    while (index > 0) {
      let parentIdx = Math.floor((index - 1) / 2);
      if (this.compare(this.heap[index], this.heap[parentIdx]) > 0) {
        [this.heap[index], this.heap[parentIdx]] = [this.heap[parentIdx], this.heap[index]];
        index = parentIdx;
      } else {
        break;
      }
    }
  }

  dequeue() {
    if (this.heap.length === 0) return null;
    if (this.heap.length === 1) return this.heap.pop();
    
    const max = this.heap[0];
    this.heap[0] = this.heap.pop();
    this.sinkDown(0);
    return max;
  }

  sinkDown(index) {
    const length = this.heap.length;
    while (true) {
      let leftIdx = 2 * index + 1;
      let rightIdx = 2 * index + 2;
      let largest = index;

      if (leftIdx < length && this.compare(this.heap[leftIdx], this.heap[largest]) > 0) {
        largest = leftIdx;
      }
      if (rightIdx < length && this.compare(this.heap[rightIdx], this.heap[largest]) > 0) {
        largest = rightIdx;
      }

      if (largest !== index) {
        [this.heap[index], this.heap[largest]] = [this.heap[largest], this.heap[index]];
        index = largest;
      } else {
        break;
      }
    }
  }

  compare(a, b) {
    // Primary: Priority level (P1 highest, P4 lowest)
    const prioMap = { P1: 4, P2: 3, P3: 2, P4: 1 };
    const scoreA = prioMap[a.urgency] || 1;
    const scoreB = prioMap[b.urgency] || 1;

    if (scoreA !== scoreB) {
      return scoreA - scoreB;
    }
    // Secondary: Earlier timestamp wins (FIFO tie-breaker)
    return new Date(b.timestamp) - new Date(a.timestamp);
  }

  size() {
    return this.heap.length;
  }

  getArray() {
    return [...this.heap];
  }
}

// 2. Circular FIFO Queue for Standard Emergencies (P3/P4)
class CircularQueue {
  constructor(capacity = 20) {
    this.items = new Array(capacity);
    this.capacity = capacity;
    this.head = 0;
    this.tail = 0;
    this.count = 0;
  }

  enqueue(item) {
    if (this.count === this.capacity) {
      return false; // Queue full
    }
    this.items[this.tail] = item;
    this.tail = (this.tail + 1) % this.capacity;
    this.count++;
    return true;
  }

  dequeue() {
    if (this.count === 0) return null;
    const item = this.items[this.head];
    this.items[this.head] = null;
    this.head = (this.head + 1) % this.capacity;
    this.count--;
    return item;
  }

  getArray() {
    const result = [];
    for (let i = 0; i < this.count; i++) {
      let idx = (this.head + i) % this.capacity;
      if (this.items[idx]) result.push(this.items[idx]);
    }
    return result;
  }
}

                  {formErrors.location && <p className="text-xs text-red-600 mt-1">{formErrors.location}</p>}
                </div>

                <div className="grid grid-cols-2 gap-3">
                  <div>
                    <label className="block text-xs font-semibold text-slate-700 mb-1">Service Required</label>
                    <select
                      value={formData.serviceType}
                      onChange={(e) => setFormData({ ...formData, serviceType: e.target.value })}
                      className="w-full px-3 py-2 text-xs border border-slate-300 rounded-lg focus:ring-2 focus:ring-red-500 outline-none bg-white"
                    >
                      <option value="Medical">Medical (Ambulance)</option>
                      <option value="Fire">Fire & Rescue</option>
                      <option value="Police">Police Assistance</option>
                    </select>
                  </div>

                  <div>
                    <label className="block text-xs font-semibold text-slate-700 mb-1">Urgency Priority Level</label>
                    <select
                      value={formData.urgency}
                      onChange={(e) => setFormData({ ...formData, urgency: e.target.value })}
                      className="w-full px-3 py-2 text-xs border border-slate-300 rounded-lg focus:ring-2 focus:ring-red-500 outline-none bg-white font-medium"
                    >
                      <option value="P1">P1 - Critical / Life-Threatening (Heap Priority)</option>
                      <option value="P2">P2 - Serious Hazard (Heap Priority)</option>
                      <option value="P3">P3 - Moderate Standard (FIFO Queue)</option>
                      <option value="P4">P4 - Non-Critical Incident (FIFO Queue)</option>
                    </select>
                  </div>
                </div>

                <div>
                  <label className="block text-xs font-semibold text-slate-700 mb-1">Situation Description</label>
                  <textarea
                    rows={3}
                    placeholder="Provide additional emergency context..."
                    value={formData.description}
                    onChange={(e) => setFormData({ ...formData, description: e.target.value })}
                    className="w-full p-2.5 text-xs border border-slate-300 rounded-lg focus:ring-2 focus:ring-red-500 outline-none"
                  ></textarea>
                </div>

                <button
                  type="submit"
                  className="w-full py-3 bg-red-600 hover:bg-red-700 text-white font-bold text-xs rounded-lg shadow transition-all flex items-center justify-center space-x-2"
                >
                  <Plus className="w-4 h-4" />
                  <span>Register Emergency & Add to DSA Queue</span>
                </button>
              </form>
            </div>

            {/* Right Side: Quick Action & Live Call Preview Card */}
            <div className="lg:col-span-7 space-y-6">
              
              {/* Dispatch Action Header Card */}
              <div className="bg-white p-5 rounded-xl border border-slate-200 shadow-sm flex items-center justify-between">
                <div>
                  <h3 className="text-sm font-bold text-slate-900">Top Priority Dispatch Control</h3>
                  <p className="text-xs text-slate-500 mt-0.5">Executes O(log N) Heap Dequeue to process highest priority emergency next.</p>
                </div>
                <button
                  onClick={handleDispatchNext}
                  className="px-5 py-3 bg-amber-500 hover:bg-amber-600 text-slate-950 font-bold text-xs rounded-lg shadow flex items-center space-x-2 transition-all"
                >
                  <Play className="w-4 h-4 fill-current" />
                  <span>Dispatch Next Call</span>
                </button>
              </div>

              {/* Active & Pending Emergencies Table */}
              <div className="bg-white rounded-xl border border-slate-200 shadow-sm overflow-hidden">
                <div className="p-4 border-b border-slate-100 flex items-center justify-between">
                  <h3 className="text-sm font-bold text-slate-900">Active Emergency Desk</h3>
                  
                  {/* Status Quick Filter */}
                  <div className="flex items-center space-x-2">
                    <span className="text-xs text-slate-500">Filter:</span>
                    <select
                      value={statusFilter}
                      onChange={(e) => setStatusFilter(e.target.value)}
                      className="text-xs border border-slate-300 rounded px-2 py-1 bg-white outline-none"
                    >
                      <option value="ALL">All Statuses</option>
                      <option value="Pending">Pending Only</option>
                      <option value="Dispatched">Dispatched</option>
                      <option value="Resolved">Resolved</option>
                      <option value="Cancelled">Cancelled</option>
                    </select>
                  </div>
                </div>

                <div className="overflow-x-auto">
                  <table className="w-full text-left text-xs">
                    <thead className="bg-slate-50 border-b border-slate-200 text-slate-600 font-semibold">
                      <tr>
                        <th className="p-3">ID & Time</th>
                        <th className="p-3">Requester / Contact</th>
                        <th className="p-3">Service</th>
                        <th className="p-3">Urgency</th>
                        <th className="p-3">Status</th>
                        <th className="p-3 text-right">Actions</th>
                      </tr>
                    </thead>
                    <tbody className="divide-y divide-slate-100">
                      {filteredRequests.length === 0 ? (
                        <tr>
                          <td colSpan={6} className="text-center p-8 text-slate-400">
                            No emergency requests matched the filter criteria.
                          </td>
                        </tr>
                      ) : (
                        filteredRequests.map((req) => (
                          <tr key={req.id} className="hover:bg-slate-50/80 transition-colors">
                            <td className="p-3 font-mono">
                              <span className="font-bold text-slate-900">{req.id}</span>
                              <div className="text-[10px] text-slate-400">{new Date(req.timestamp).toLocaleTimeString()}</div>
                            </td>
                            <td className="p-3">
                              <div className="font-semibold text-slate-800">{req.name}</div>
                              <div className="text-[11px] text-slate-500">{req.contact}</div>
                            </td>
                            <td className="p-3">
                              <span className={`inline-flex items-center px-2 py-0.5 rounded text-[11px] font-medium ${
                                req.serviceType === 'Medical' ? 'bg-red-50 text-red-700 border border-red-200' :
                                req.serviceType === 'Fire' ? 'bg-orange-50 text-orange-700 border border-orange-200' :
                                'bg-blue-50 text-blue-700 border border-blue-200'
                              }`}>
                                {req.serviceType}
                              </span>
                            </td>
                            <td className="p-3">
                              <span className={`font-bold px-2 py-0.5 rounded text-[10px] ${
                                req.urgency === 'P1' ? 'bg-red-100 text-red-800' :
                                req.urgency === 'P2' ? 'bg-amber-100 text-amber-800' :
                                req.urgency === 'P3' ? 'bg-yellow-100 text-yellow-800' : 'bg-slate-100 text-slate-700'
                              }`}>
                                {req.urgency}
                              </span>
                            </td>
                            <td className="p-3">
                              <span className={`px-2 py-0.5 rounded text-[11px] font-semibold ${
                                req.status === 'Pending' ? 'bg-amber-100 text-amber-800' :
                                req.status === 'Dispatched' ? 'bg-blue-100 text-blue-800' :
                                req.status === 'Resolved' ? 'bg-emerald-100 text-emerald-800' : 'bg-slate-100 text-slate-600'
                              }`}>
                                {req.status}
                              </span>
                            </td>
                            <td className="p-3 text-right space-x-1">
                              {req.status === 'Pending' && (
                                <button
                                  onClick={() => handleStatusChange(req.id, 'Dispatched')}
                                  title="Dispatch"
                                  className="p-1 bg-blue-50 text-blue-600 hover:bg-blue-100 rounded"
                                >
                                  <Play className="w-3.5 h-3.5" />
                                </button>
                              )}
                              {req.status === 'Dispatched' && (
                                <button
                                  onClick={() => handleStatusChange(req.id, 'Resolved')}
                                  title="Mark Resolved"
                                  className="p-1 bg-emerald-50 text-emerald-600 hover:bg-emerald-100 rounded"
                                >
                                  <Check className="w-3.5 h-3.5" />
                                </button>
                              )}
                              <button
                                onClick={() => setSelectedEmergency(req)}
                                title="View Details"
                                className="p-1 bg-slate-100 text-slate-700 hover:bg-slate-200 rounded"
                              >
                                <FileText className="w-3.5 h-3.5" />
                              </button>
                              <button
                                onClick={() => handleDelete(req.id)}
                                title="Delete Record"
                                className="p-1 bg-red-50 text-red-600 hover:bg-red-100 rounded"
                              >
                                <Trash2 className="w-3.5 h-3.5" />
                              </button>
                            </td>
                          </tr>
                        ))
                      )}
                    </tbody>
                  </table>
                </div>
              </div>

            </div>

          </div>
        )}

        {/* TAB 2: LIVE QUEUE VISUALIZER */}
        {activeTab === 'queues' && (
          <div className="space-y-6">
            
            {/* Priority Queue Heap Visualization Card */}
            <div className="bg-white rounded-xl border border-slate-200 p-6 shadow-sm">
              <div className="flex items-center justify-between pb-4 border-b border-slate-100 mb-4">
                <div>
                  <h3 className="text-base font-bold text-slate-900 flex items-center space-x-2">
                    <span className="w-2.5 h-2.5 rounded-full bg-red-600"></span>
                    <span>1. Priority Queue Visualizer (Binary Max-Heap: P1 & P2 Calls)</span>
                  </h3>
                  <p className="text-xs text-slate-500 mt-1">High-urgency emergency calls arranged automatically by heap priority.</p>
                </div>
                <span className="text-xs font-mono font-bold px-3 py-1 bg-red-50 text-red-700 rounded-full border border-red-200">
                  Heap Count: {priorityQueueArray.length}
                </span>
              </div>

              {priorityQueueArray.length === 0 ? (
                <div className="p-8 text-center text-slate-400 bg-slate-50/50 rounded-lg border border-dashed border-slate-200">
                  No high-priority calls in the Binary Max-Heap queue.
                </div>
              ) : (
                <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
                  {priorityQueueArray.map((item, idx) => (
                    <div 
                      key={item.id} 
                      className={`p-4 rounded-lg border relative transition-all ${
                        idx === 0 
                          ? 'border-red-400 bg-red-50/30 ring-2 ring-red-500 ring-offset-1' 
                          : 'border-slate-200 bg-white'
                      }`}
                    >
                      {idx === 0 && (
                        <span className="absolute -top-2.5 left-3 px-2 py-0.5 bg-red-600 text-white font-bold text-[10px] rounded shadow">
                          FRONT OF HEAP QUEUE
                        </span>
                      )}
                      <div className="flex items-center justify-between mb-2">
                        <span className="font-mono font-bold text-slate-900 text-xs">{item.id}</span>
                        <span className="px-2 py-0.5 bg-red-100 text-red-800 text-[10px] font-bold rounded">
                          {item.urgency}
                        </span>
                      </div>
                      <p className="text-xs font-semibold text-slate-800">{item.name}</p>
                      <p className="text-xs text-slate-500">{item.location}</p>
                      <div className="mt-3 pt-2 border-t border-slate-100 flex items-center justify-between text-[11px] text-slate-400">
                        <span>{item.serviceType}</span>
                        <span>{new Date(item.timestamp).toLocaleTimeString()}</span>
                      </div>
                    </div>
                  ))}
                </div>
              )}
            </div>

            {/* Circular Standard Queue Visualization Card */}
            <div className="bg-white rounded-xl border border-slate-200 p-6 shadow-sm">
              <div className="flex items-center justify-between pb-4 border-b border-slate-100 mb-4">
                <div>
                  <h3 className="text-base font-bold text-slate-900 flex items-center space-x-2">
                    <span className="w-2.5 h-2.5 rounded-full bg-blue-600"></span>
                    <span>2. Standard Queue Visualizer (Circular Queue: P3 & P4 Calls)</span>
                  </h3>
                  <p className="text-xs text-slate-500 mt-1">Standard emergencies processed in First-In, First-Out (FIFO) chronological sequence.</p>
                </div>
                <span className="text-xs font-mono font-bold px-3 py-1 bg-blue-50 text-blue-700 rounded-full border border-blue-200">
                  Standard Count: {standardQueueArray.length}
                </span>
              </div>

              {standardQueueArray.length === 0 ? (
                <div className="p-8 text-center text-slate-400 bg-slate-50/50 rounded-lg border border-dashed border-slate-200">
                  Standard circular queue is empty.
                </div>
              ) : (
                <div className="flex items-center space-x-3 overflow-x-auto pb-2">
                  {standardQueueArray.map((item, idx) => (
                    <div 
                      key={item.id}
                      className="min-w-[220px] p-4 rounded-lg border border-slate-200 bg-white shrink-0 shadow-sm"
                    >
                      <div className="flex items-center justify-between mb-1">
                        <span className="font-mono font-bold text-xs text-slate-900">{item.id}</span>
                        <span className="px-1.5 py-0.5 bg-yellow-100 text-yellow-800 font-bold text-[10px] rounded">
                          {item.urgency}
                        </span>
                      </div>
                      <p className="text-xs font-semibold text-slate-800">{item.name}</p>
                      <p className="text-[11px] text-slate-500 truncate">{item.location}</p>
                      <span className="mt-2 block text-[10px] text-slate-400 font-mono">
                        FIFO Index: #{idx + 1}
                      </span>
                    </div>
                  ))}
                </div>
              )}
            </div>

          </div>
        )}

        {/* TAB 3: REPORTS & ANALYTICS */}
        {activeTab === 'reports' && (
          <div className="bg-white rounded-xl border border-slate-200 p-6 shadow-sm space-y-6">
            
            <div className="flex items-center justify-between pb-4 border-b border-slate-100">
              <div>
                <h3 className="text-base font-bold text-slate-900">Emergency Incident History & Official Export</h3>
                <p className="text-xs text-slate-500">Filter, query, and download complete incident audit logs.</p>
              </div>

              {/* PDF Download Trigger */}
              <button
                onClick={generateAndDownloadPDF}
                className="px-4 py-2 bg-red-600 hover:bg-red-700 text-white font-bold text-xs rounded-lg shadow flex items-center space-x-2 transition-all"
              >
                <Download className="w-4 h-4" />
                <span>Download PDF File Report</span>
              </button>
            </div>

            {/* Instant Search Bar */}
            <div className="relative">
              <Search className="w-4 h-4 text-slate-400 absolute left-3 top-3" />
              <input
                type="text"
                placeholder="Search emergency by ID, Requester Name, Location, or Contact..."
                value={searchQuery}
                onChange={(e) => setSearchQuery(e.target.value)}
                className="w-full pl-9 pr-3 py-2 text-xs border border-slate-300 rounded-lg outline-none focus:ring-2 focus:ring-red-500"
              />
            </div>

            {/* Complete Data Table */}
            <div className="overflow-x-auto border border-slate-200 rounded-lg">
              <table className="w-full text-left text-xs">
                <thead className="bg-slate-100 border-b border-slate-200 text-slate-700 font-bold">
                  <tr>
                    <th className="p-3">Ticket ID</th>
                    <th className="p-3">Requester Name</th>
                    <th className="p-3">Contact</th>
                    <th className="p-3">Service</th>
                    <th className="p-3">Urgency</th>
                    <th className="p-3">Location</th>
                    <th className="p-3">Status</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-slate-100">
                  {filteredRequests.map(req => (
                    <tr key={req.id} className="hover:bg-slate-50">
                      <td className="p-3 font-mono font-bold text-slate-900">{req.id}</td>
                      <td className="p-3 font-medium text-slate-800">{req.name}</td>
                      <td className="p-3 text-slate-600">{req.contact}</td>
                      <td className="p-3">{req.serviceType}</td>
                      <td className="p-3 font-bold">{req.urgency}</td>
                      <td className="p-3 text-slate-600">{req.location}</td>
                      <td className="p-3">
                        <span className={`px-2 py-0.5 rounded text-[10px] font-bold ${
                          req.status === 'Pending' ? 'bg-amber-100 text-amber-800' :
                          req.status === 'Dispatched' ? 'bg-blue-100 text-blue-800' :
                          req.status === 'Resolved' ? 'bg-emerald-100 text-emerald-800' : 'bg-slate-100 text-slate-600'
                        }`}>
                          {req.status}
                        </span>
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>

          </div>
        )}

      </main>

      {/* Emergency Detail Modal View */}
      {selectedEmergency && (
        <div className="fixed inset-0 z-50 bg-slate-900/40 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="bg-white rounded-xl border border-slate-200 p-6 max-w-md w-full shadow-2xl relative">
            <button
              onClick={() => setSelectedEmergency(null)}
              className="absolute top-4 right-4 text-slate-400 hover:text-slate-600"
            >
              <X className="w-5 h-5" />
            </button>

            <div className="flex items-center space-x-2 mb-4 pb-3 border-b border-slate-100">
              <ShieldAlert className="w-5 h-5 text-red-600" />
              <h3 className="text-sm font-bold text-slate-900">Incident Ticket: {selectedEmergency.id}</h3>
            </div>

            <div className="space-y-3 text-xs">
              <div>
                <span className="text-slate-400">Requester:</span>
                <p className="font-semibold text-slate-800">{selectedEmergency.name}</p>
              </div>
              <div>
                <span className="text-slate-400">Contact:</span>
                <p className="font-semibold text-slate-800">{selectedEmergency.contact}</p>
              </div>
              <div>
                <span className="text-slate-400">Location:</span>
                <p className="font-semibold text-slate-800">{selectedEmergency.location}</p>
              </div>
              <div>
                <span className="text-slate-400">Details / Caller Context:</span>
                <p className="p-2.5 bg-slate-50 border border-slate-200 rounded mt-1 text-slate-700">{selectedEmergency.description}</p>
              </div>
            </div>

            <div className="mt-6 pt-3 border-t border-slate-100 flex justify-end">
              <button
                onClick={() => setSelectedEmergency(null)}
                className="px-4 py-2 bg-slate-800 text-white rounded text-xs font-semibold"
              >
                Close View
              </button>
            </div>
          </div>
        </div>
      )}

    </div>
  );
}
